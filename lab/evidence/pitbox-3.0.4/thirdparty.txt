== lin/Pitbox/instalar.sh
#!/usr/bin/env bash
# Prepara Pitbox (sin sudo) y lo abre. Necesita: python3, python3-venv, python3-gi y gir1.2-webkit2-4.1.
set -euo pipefail
cd "$(dirname "$0")"
python3 -m venv --system-site-packages .venv
.venv/bin/pip install -q -r requirements.txt
exec .venv/bin/python gpbmods_app.py
(exit 0)
