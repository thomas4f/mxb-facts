def _registro(clave: str, valor: str):
    try:
        import winreg

        with winreg.OpenKey(winreg.HKEY_CURRENT_USER, clave) as k:
            return winreg.QueryValueEx(k, valor)[0]
    except (ImportError, OSError):
        return None

def _windows_registro_steam() -> tuple[Path | None, Path | None]:
    try:
        import winreg

        with winreg.OpenKey(winreg.HKEY_CURRENT_USER, r"Software\Valve\Steam") as k:
            ruta = winreg.QueryValueEx(k, "SteamPath")[0]
            try:
                exe = winreg.QueryValueEx(k, "SteamExe")[0]
            except OSError:
                exe = str(Path(ruta) / "steam.exe")
        return Path(ruta), Path(exe)
