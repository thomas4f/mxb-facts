def _cat_en_pkz_cifrado(pkz: Path) -> str | None:
    """Los .pkz de mods van cifrados, pero muchos guardan el .ini legible al final del archivo."""
    with open(pkz, "rb") as f:
        f.seek(0, 2)
        f.seek(max(0, f.tell() - (1 << 20)))
        cola = f.read()
    m = _CAT_RE.search(cola)
    return m.group(1).decode("latin-1").strip() if m else None
