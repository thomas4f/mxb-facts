layout(location=0) in vec3 aPos;
layout(location=1) in vec2 aSizeRot;      // size, rotation (one record per particle: instanced)
layout(location=2) in vec3 aVel;
layout(location=3) in vec4 aColor;
layout(location=4) in vec4 aParam;        // kind, life 0..1, seed, heat
uniform mat4 uView, uProj;
uniform sampler2D uScene; uniform float uMaxLod;
out vec2 vCorner; out vec4 vColor; out vec4 vParam; out float vZ; out float vStretch;
flat out vec3 vSky, vGnd; flat out float vEmis;
const vec2 C[6] = vec2[6](vec2(-1.0,-1.0), vec2(1.0,-1.0), vec2(1.0,1.0), vec2(-1.0,-1.0), vec2(1.0,1.0), vec2(-1.0,1.0));
void main(){
  // the same for every pixel: fetched once per vertex instead of per pixel
  vSky = textureLod(uScene, vec2(0.5, 0.86), uMaxLod - 2.0).rgb;
  vGnd = textureLod(uScene, vec2(0.5, 0.12), uMaxLod - 2.0).rgb;
  vEmis = 5.0 + 14.0 * clamp(dot(textureLod(uScene, vec2(0.5), uMaxLod).rgb, vec3(0.2126, 0.7152, 0.0722)), 0.0, 4.0);
  vec4 c = uView * vec4(aPos, 1.0);
  int kind = int(aParam.x + 0.5);
  vec2 k = C[gl_VertexID]; float size = aSizeRot.x;
  vec2 off; vStretch = 1.0;
  if (kind == 9) {
    // lightning: a ribbon from aPos to aPos + aVel, facing the camera
    vec4 cb = uView * vec4(aPos + aVel, 1.0);
    vec2 dd = cb.xy / max(-cb.z, 1.0) - c.xy / max(-c.z, 1.0);
    vec2 ax = length(dd) > 1e-6 ? normalize(dd) : vec2(0.0, 1.0); vec2 ay = vec2(-ax.y, ax.x);
    vec4 e = k.x < 0.0 ? c : cb;
    e.xy += ay * k.y * size * 6.0 + ax * k.x * size * 3.0;
    gl_Position = uProj * e;
    vCorner = k; vColor = aColor; vParam = aParam; vZ = -e.z; vStretch = 1.0;
    return;
  if (kind == 2 || kind == 3 || kind == 8) {
    vec3 vv = mat3(uView) * aVel;
    vec2 d = vv.xy / max(-c.z, 0.3);
    float L = length(d);
    vec2 ax = L > 1e-5 ? d / L : vec2(1.0, 0.0); vec2 ay = vec2(-ax.y, ax.x);
    float streak = kind == 3 ? 0.016 : kind == 8 ? 0.008 : 0.03;
    float along = size + L * max(-c.z, 0.3) * streak;
    vStretch = along / size;
    off = ax * (k.x * along - (kind == 3 || kind == 8 ? along * 0.8 : along * 0.5)) + ay * k.y * size * (kind == 3 ? 0.25 : kind == 8 ? 0.5 : 0.8);
  } else {
    float cr = cos(aSizeRot.y), sr = sin(aSizeRot.y);
    off = vec2(cr * k.x - sr * k.y, sr * k.x + cr * k.y) * size;
  c.xy += off;
  gl_Position = uProj * c;
  vCorner = k; vColor = aColor; vParam = aParam; vZ = -c.z;
#version 330
in vec2 vCorner; in vec4 vColor; in vec4 vParam; in float vZ; in float vStretch;
flat in vec3 vSky, vGnd; flat in float vEmis;
uniform sampler2D uScene;   // HDR scene so far (mipmapped)
uniform sampler2D uDepth;   // scene depth copy
uniform sampler2D uFull;    // full-size scene copy (the haze bends it)
uniform sampler3D uNoise;   // baked tiling fbm: r 4 octaves, g 3 finer ones
uniform vec4 uVp;           // viewport x, y, w, h
uniform vec2 uDepthP;       // projection P[10], P[14]
uniform float uTime, uHasDepth, uLod, uMaxLod;
uniform int uPass;
uniform vec4 uSparkCol;    // spark hue / saturation / brightness from the menu
layout(location=0) out vec4 o;
float h3(vec3 p){ p = fract(p * 0.3183099 + 0.1); p *= 17.0; return fract(p.x * p.y * p.z * (p.x + p.y + p.z)); }
float n3(vec3 x){ vec3 i = floor(x), f = fract(x); f = f * f * (3.0 - 2.0 * f);
  return mix(mix(mix(h3(i), h3(i + vec3(1,0,0)), f.x), mix(h3(i + vec3(0,1,0)), h3(i + vec3(1,1,0)), f.x), f.y),
             mix(mix(h3(i + vec3(0,0,1)), h3(i + vec3(1,0,1)), f.x), mix(h3(i + vec3(0,1,1)), h3(i + vec3(1,1,1)), f.x), f.y), f.z); }
float fbm(vec3 p){ float a = 0.5, s = 0.0; for (int i = 0; i < 4; i++){ s += a * n3(p); p = p * 2.07 + vec3(1.7, 9.2, 3.1); a *= 0.5; } return s; }
float fbm3(vec3 p){ return n3(p) * 0.57 + n3(p * 2.07 + vec3(1.7, 9.2, 3.1)) * 0.29 + n3(p * 4.28 + vec3(5.2, 1.1, 7.7)) * 0.14; }
vec3 fire(float T){
  vec3 c = mix(vec3(0.35, 0.02, 0.0), vec3(1.0, 0.28, 0.03), smoothstep(0.0, 0.35, T));
  c = mix(c, vec3(1.0, 0.62, 0.16), smoothstep(0.3, 0.6, T));
  c = mix(c, vec3(1.0, 0.9, 0.62), smoothstep(0.55, 0.82, T));
  return mix(c, vec3(1.0, 0.96, 0.86), smoothstep(0.84, 1.0, T));   // white-hot, never blue
void main(){
  int kind = int(vParam.x + 0.5); float life = vParam.y, seed = vParam.z, heat = vParam.w;
  // emissive scale follows the scene's exposure, so fire reads bright by day and by night
  float uEmis = vEmis;
  if ((uPass == 1) != (kind == 2 || kind == 3 || kind == 4 || kind == 9)) discard;
  vec2 q = vCorner; float r = length(q);
  if (r > 1.0 && kind != 9) discard;

  float soft = 1.0;
  if (uHasDepth > 0.5) {
    float d = texture(uDepth, suv).r * 2.0 - 1.0;
    float sz = uDepthP.y / (d + uDepthP.x);
    soft = clamp((sz - vZ) / (kind == 5 ? 0.05 : kind == 1 ? 0.25 : 0.45), 0.0, 1.0);
  float fadeIn = smoothstep(0.0, 0.06, life), fadeOut = 1.0 - smoothstep(0.45, 1.0, life);
  if (kind == 0 || kind == 6) {

    float t = uTime * 0.28 + seed;

    float n = texture(uNoise, p * (1.5 / 4.0)).r;
    float n2 = texture(uNoise, (p * 3.4 + 7.0) / 8.0).g;
    float edge = smoothstep(1.0, 0.25, r);
    float dens = clamp((n * 0.85 + n2 * 0.3 - 0.3 - r * r * 0.5) * 2.3, 0.0, 1.0) * edge;

    vec3 gnd = vGnd;

    vec3 light = hemi * 1.1 + amb * 0.2 + vec3(skyL) * (0.6 + 0.4 * N.y);   // + sun, neutral white
    light += sky * 0.35 * pow(1.0 - abs(N.z), 2.0);   // bright rim where it thins

    vec3 col = vColor.rgb * light * (1.0 - 0.38 * thick);

    o = vec4(col * a, a);

    vec3 c = textureLod(uFull, uv2, 0.0).rgb;
    float a = clamp(s * 1.6, 0.0, 1.0);
    o = vec4(c * a, a);
  } else if (kind == 2) {
    // flame tongue: wide and white-hot at the root (q.x = -1, toward the pipe), narrowing to a
    // sharp ragged point (q.x = 1), wiggling along its length and flickering

    float ts = uTime * 9.0 + seed * 7.0;
    float wig = (texture(uNoise, vec3(x * 0.35 + seed * 1.7, seed * 2.3, ts * 0.03)).r - 0.5) * 1.4 * x;

    float rid = 1.0 - abs(n1 * 2.0 - 1.0);                         // ridges -> sharp fingers

    float body = smoothstep(0.0, 0.06, edge);                        // crisp edge, not a soft blob
    float core = smoothstep(0.08, 0.5, edge) * (1.0 - 0.8 * x);
    float T = clamp(heat * (1.0 - life * 0.55) * (0.38 + 0.8 * core - 0.25 * x), 0.0, 1.0);

    I *= 0.4 + 0.6 * smoothstep(0.08, 0.4, T);

  } else if (kind == 3) {

    float across = exp(-q.y * q.y * 9.0);
    float alongF = smoothstep(-1.0, 0.6, q.x) * smoothstep(1.0, 0.75, q.x);

    vec3 hc = clamp(abs(fract(uSparkCol.x + vec3(0.0, 2.0/3.0, 1.0/3.0)) * 6.0 - 3.0) - 1.0, 0.0, 1.0);

float luma(vec3 c){ return dot(c, vec3(0.2126, 0.7152, 0.0722)); }
// hash from the float bits: no period (the old fract() hash repeated every ~10 cells: tiled snow)
float h2(vec2 p){ uvec2 q = floatBitsToUint(p); uint h = q.x * 0x8da6b343u ^ q.y * 0xd8163841u ^ 0x9e3779b9u;
  h ^= h >> 15; h *= 0x2c1b3c6du; h ^= h >> 12; h *= 0x297a2d39u; h ^= h >> 15; return float(h & 0xffffffu) / 16777216.0; }
float vn(vec2 p){ vec2 i = floor(p), f = fract(p); f = f * f * (3.0 - 2.0 * f);
  return mix(mix(h2(i), h2(i + vec2(1,0)), f.x), mix(h2(i + vec2(0,1)), h2(i + vec2(1,1)), f.x), f.y); }
float fb(vec2 p){ return vn(p) * 0.55 + vn(p * 2.13 + 7.1) * 0.3 + vn(p * 4.7 + 3.3) * 0.15; }
// anti-aliased noise: each layer fades to its average once it's finer than a pixel on the ground
// (far away, at a shallow angle, a pixel covers metres of track: the sub-pixel TAA jitter then hit a
// different bit of the pattern every frame - the snow, leaves and autumn colours crawled)
float gFp = 0.0;
float aaK(float f){ return 1.0 - smoothstep(0.2, 0.8, gFp * f); }
float vnA(vec2 q, float f){ return mix(0.5, vn(q), aaK(f)); }
float fbA(vec2 q, float f){ return vnA(q, f) * 0.55 + vnA(q * 2.13 + 7.1, f * 2.13) * 0.3 + vnA(q * 4.7 + 3.3, f * 4.7) * 0.15; }
vec3 autumnPal(float h){
  vec3 a = vec3(0.86, 0.36, 0.06), r = vec3(0.68, 0.12, 0.04), y = vec3(0.92, 0.66, 0.12), b = vec3(0.45, 0.25, 0.08), g = vec3(0.30, 0.42, 0.10);
  return h < 0.28 ? a : h < 0.45 ? r : h < 0.68 ? y : h < 0.86 ? b : g; }
vec3 hue(float t){ return clamp(abs(fract(t + vec3(0.0, 2.0/3.0, 1.0/3.0)) * 6.0 - 3.0) - 1.0, 0.0, 1.0); }

  uCam = uCam4.xyz; uSun = uSun4.xyz;
  ivec2 ip = ivec2(gl_FragCoord.xy);
  vec3 col = texelFetch(uScene, ip - ivec2(uVp.xy), 0).rgb;
  float d = uB.z > 0.5 ? texelFetch(uDepth, ip - ivec2(uVp.xy), 0).r : 1.0;
  vec2 uv = (gl_FragCoord.xy - uVp.xy) / uVp.zw;
  vec4 w = uInvVP * vec4(uv * 2.0 - 1.0, d * 2.0 - 1.0, 1.0); vec3 P = w.xyz / w.w;
  vec4 wf = uInvVP * vec4(uv * 2.0 - 1.0, 1.0, 1.0);
  vec3 ray = normalize(wf.xyz / wf.w - uCam);
  bool sky = d >= 0.999999 && ray.y > -0.012;
  bool glass = d >= 0.999999 && !sky;     // seen through the windscreen (no depth)
  float dist = sky || glass ? 1e5 : length(P - uCam);
  gFp = sky ? 1e3 : min(max(length(dFdx(P.xz)), length(dFdy(P.xz))), 50.0);
  vec3 N = normalize(cross(dFdx(P), dFdy(P))); if (dot(N, ray) > 0.0) N = -N;
  if (sky || glass) N = vec3(0.0, 1.0, 0.0);
  float mx = max(col.r, max(col.g, col.b)), mn = min(col.r, min(col.g, col.b));
  float sat = (mx - mn) / max(mx, 1e-4), lum = luma(col);
  float green = (sky || glass) ? 0.0 : smoothstep(0.04, 0.16, (col.g - max(col.r, col.b)) / max(mx, 1e-4)) * smoothstep(0.004, 0.02, lum);
  // painted things (tyre walls, boards, kerbs) are flat, strongly coloured and smooth - foliage is
  // speckled: keep the paint out of the autumn / snow colouring
  float dl = (abs(dFdx(lum)) + abs(dFdy(lum))) / max(lum, 1e-3);
  green *= 1.0 - smoothstep(0.45, 0.7, sat) * (1.0 - smoothstep(0.04, 0.14, dl));
  float up = smoothstep(0.55, 0.85, N.y);
  vec3 res = col;
  float mark = 0.0;     // how much of this pixel the final grade must keep (over the sky)
  // what the ground is: asphalt (dark, grey) or the rest (grass, sand, gravel) - from the blurred copy,
  // so the texture's own pixels don't make a mosaic; bikes and riders are never ground
  // (performance: none of this unless a season effect is on)
  bool seasonsOn = uA.x > 0.0 || uA.y > 0.0 || uA.z > 0.0 || uSnowT > 0.0;
  vec3 cl = seasonsOn ? textureLod(uLow, uv, 2.5).rgb : vec3(0.5); float mxl = max(cl.r, max(cl.g, cl.b)), mnl = min(cl.r, min(cl.g, cl.b));
  float satL = (mxl - mnl) / max(mxl, 1e-4), lumL = luma(cl);
  float greenL = smoothstep(0.02, 0.12, (cl.g - max(cl.r, cl.b)) / max(mxl, 1e-4));
  // bikes and riders (and, close to the camera, your own bike, dash and hands) are never ground or
  // foliage: anything within ~1.7 m of a bike that isn't the flat track surface under it
  bool nearBike = dist < 1.4;
  if (seasonsOn) for (int i = 0; i < 8; i++) { if (float(i) >= uBkN) break;
    vec3 dbk = P - uBk[i].xyz; float gY = min(uBk[i].w, uBk[i].y - 0.25);
    // the bike and rider: near it, and either not facing up or above the ground under it (the bike's
    // reference point is around its middle) - the track under the bike keeps its snow
    if (length(dbk) < 1.7 && (N.y < 0.75 || P.y > uBk[i].y - 0.15)) nearBike = true; }
  float ground = (sky || glass || nearBike) ? 0.0 : up;
  // asphalt: grey (any brightness - sunlit or in shadow); sand and gravel are beige, grass green
  float asph = ground * (1.0 - smoothstep(0.08, 0.16, satL));
  float detailK = 1.0 - smoothstep(6.0, 30.0, dist);   // fine detail only up close (it shimmered far away)
  float rest = ground * (1.0 - asph);
  // autumn: green leaves and grass turn orange, red, yellow and brown, tree by tree
  if (uA.x > 0.0 && green > 0.0 && !nearBike) {

    float fine = vnA(P.xz * 1.7 + P.y * 1.3, 1.7);

    vec3 tgt = mix(pal, dry, up * 0.7);
    tgt = tgt / max(luma(tgt), 1e-3) * lum * mix(1.05, 0.95, up);
    res = mix(res, tgt, green * uA.x * mix(1.0, 0.6, up));
  // snow on grass and trees: the trees' leaves, and all the ground that isn't the track (grass, sand,
  // gravel run-off), lit as the scene is

    float clump = fbA(P.xz * 1.3 + P.y * 1.1, 1.3);

    float gS = mix(greenL, green, 0.45);   // the leaves, steadied by the blurred copy (grass blades speckled)
